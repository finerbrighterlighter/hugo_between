Moving institution is an edit in data/homepage.yml and nowhere else. */ -}} {{- $sb := .sidebar -}} {{- $parts := slice -}} {{- with $sb.faculty }}{{ $parts = $parts | append . }}{{ end -}} {{- with $sb.university }}{{ $parts = $parts | append . }}{{ end -}} {{- with $sb.city }}{{ $parts = $parts | append . }}{{ end -}} {{- return (delimit $parts ", ") -}}
