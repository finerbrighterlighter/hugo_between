{{- /* Who the work was done with, by how often. The sentence above counts the collaborators and the institutions. This is the shape behind that number, and it is the shape research collaboration actually has: a handful of people on nearly everything, then a long tail of one-time coauthors. A total alone hides which of those it is, and the tail is not padding — it is most of the names. No caption: the sentence above is the caption. The names and counts are text, so the bars carry nothing the reader cannot already read and are hidden from assistive technology. The tail folds away the way the chronology's does, so the list keeps its height as the work grows. */ -}} {{- $enSite := site -}} {{- range hugo.Sites -}}{{- if eq .Language.Lang "en" -}}{{- $enSite = . -}}{{- end -}}{{- end -}} {{- $rm := dict -}} {{- range hugo.Data.researchers -}}{{- $rm = merge $rm (dict .id .) -}}{{- end -}} {{- $counts := dict -}} {{- range where $enSite.RegularPages "Section" "works" -}} {{- $seen := slice -}} {{- range .Params.authors -}} {{- $id := .id -}} {{- if and $id (not .highlight) (not (in $seen $id)) -}} {{- $seen = $seen | append $id -}} {{- $counts = merge $counts (dict $id (add (index $counts $id | default 0) 1)) -}} {{- end -}} {{- end -}} {{- end -}} {{- $rows := slice -}} {{- range $id, $n := $counts -}} {{- $name := $id -}} {{- with index $rm $id -}} {{- $initials := "" -}} {{- range split (strings.TrimSpace (.given | default "")) " " -}} {{- with . -}}{{- $initials = printf "%s%s" $initials (substr . 0 1) -}}{{- end -}} {{- end -}} {{- $name = strings.TrimSpace (printf "%s %s" .family $initials) -}} {{- end -}} {{- $rows = $rows | append (dict "name" $name "n" $n) -}} {{- end -}} {{- with $rows -}} {{- $rows = sort (sort $rows "name" "asc") "n" "desc" -}} {{- $max := (index $rows 0).n -}} {{- $limit := site.Params.collabVisibleLimit | default 6 -}} {{- $head := $rows -}}{{- $tail := slice -}} {{- if gt (len $rows) $limit -}} {{- $head = first $limit $rows -}}{{- $tail = after $limit $rows -}} {{- end -}} {{- end -}}
