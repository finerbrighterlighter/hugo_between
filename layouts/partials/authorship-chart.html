{{- /* Research output, read as author lists. Replaces the separate output timeline, which plotted one mark per work by year. This does that — a column per work, grouped under its year, so how much came out of a year is how many columns it holds — and adds the thing a publication list never says: which of them were mine to lead. Note that a year is not taller for being more productive: every group is the height of the longest author list in the whole figure, because the year labels have to sit on one line. Volume reads across, not up. One slot per author, first at the top, my slot marked. Sole author is a single mark; third of eleven is a long column with the mark near the top. The old figure's one encoding is kept: filled for written output, open for presented. It is carried by my slot rather than the whole column. Works are spaced evenly inside a year rather than by date; four of them fall within months of each other in 2023 and would collide. HTML and CSS rather than SVG, so the year labels stay prose-sized text and the columns reflow — the old timeline had to be hidden below 700px. */ -}} {{- $enSite := site -}} {{- range hugo.Sites -}}{{- if eq .Language.Lang "en" -}}{{- $enSite = . -}}{{- end -}}{{- end -}} {{- $presented := slice "conference-speaking" "conference-poster" -}} {{- $pages := where (where $enSite.RegularPages "Section" "works") "Params.type" "ne" "under-review" -}} {{- $listURL := (partial "lang-link.html" "/works/").href -}} {{- /* Above this many authors the column stops counting and starts being a wall of dots, so it is truncated with a final marker slot. The tip and the accessible name always carry the true position and total. */ -}} {{- $cap := 20 -}} {{- $works := slice -}} {{- range (sort $pages "Date" "asc") -}} {{- /* `default` cannot rescue this: len runs first and dies on a missing key with "reflect: call of reflect.Value.Type on zero Value", taking the whole build with it. An author entry may also be a bare string. */ -}} {{- $authors := .Params.authors | default slice -}} {{- $n := len $authors -}} {{- $pos := 0 -}} {{- range $i, $a := $authors -}} {{- if and (reflect.IsMap $a) $a.highlight -}}{{- $pos = add $i 1 -}}{{- end -}} {{- end -}} {{- /* No author list at all: one slot, and it is mine. An author list with nobody flagged is a different thing — the work is not sole-authored, so the column is drawn with no mark rather than inventing one. */ -}} {{- if eq $n 0 -}}{{- $n = 1 -}}{{- $pos = 1 -}} {{- else if eq $pos 0 -}} {{- warnf "authorship: %q has %d authors and none flagged `highlight: true`; drawn without a mark" .Title $n -}} {{- end -}} {{- /* conference items and reports are list-only, so those fall back to the works list filtered to the year, never to a publisher or a mirror */ -}} {{- $href := .RelPermalink -}} {{- if not $href -}}{{- $href = printf "%s?search=%d" $listURL .Date.Year -}}{{- end -}} {{- $works = $works | append (dict "title" .Title "year" .Date.Year "pos" $pos "n" $n "href" $href "slots" (cond (gt $n $cap) $cap $n) "mark" (cond (gt $pos $cap) $cap $pos) "clipped" (gt $n $cap) "open" (in $presented .Params.type) "kind" (i18n (printf "type_%s" .Params.type) | default .Params.type)) -}} {{- end -}} {{- with $works -}} {{- $years := slice -}} {{- range $works -}}{{- if not (in $years .year) -}}{{- $years = $years | append .year -}}{{- end -}}{{- end -}} {{- /* the longest author list sets every year group's height, so the year labels sit on one line instead of stepping with each group */ -}} {{- $maxN := 1 -}} {{- range $works -}}{{- if gt .slots $maxN -}}{{- $maxN = .slots -}}{{- end -}}{{- end -}} {{- end -}}
