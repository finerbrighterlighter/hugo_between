{{- /* Research output, read as author lists. Replaces the separate output timeline, which plotted one mark per work by year. This does that — a column per work, grouped under its year, so how much came out of a year is how many columns it holds — and adds the thing a publication list never says: which of them were mine to lead. Three rungs, not one slot per author. First, middle, last is what an author list actually encodes: the ends carry the claim and the middle is a crowd. Drawing every author instead made the figure mostly grey, made one consortium paper set the height of the whole thing, and implied that being 27th of 60 differs from being 31st, which it does not. A sole author is a first author and nothing else; marking the last rung too would claim a seniority position that a one-name paper does not have. A fourth rung, a diamond, is the corresponding author: a role rather than a position, since it can fall anywhere in the list. Set `corresponding: true` on the author entry that carries `highlight`. The old figure's one encoding is kept: filled for written output, open for presented. It is carried by my slot rather than the whole column. Works are spaced evenly inside a year rather than by date; four of them fall within months of each other in 2023 and would collide. HTML and CSS rather than SVG, so the year labels stay prose-sized text and the columns reflow — the old timeline had to be hidden below 700px. */ -}} {{- $enSite := site -}} {{- range hugo.Sites -}}{{- if eq .Language.Lang "en" -}}{{- $enSite = . -}}{{- end -}}{{- end -}} {{- $presented := slice "conference-speaking" "conference-poster" -}} {{- $pages := where (where $enSite.RegularPages "Section" "works") "Params.type" "ne" "under-review" -}} {{- $listURL := (partial "lang-link.html" "/works/").href -}} {{- $works := slice -}} {{- range (sort $pages "Date" "asc") -}} {{- /* `default` cannot rescue this: len runs first and dies on a missing key with "reflect: call of reflect.Value.Type on zero Value", taking the whole build with it. An author entry may also be a bare string. */ -}} {{- $authors := .Params.authors | default slice -}} {{- $n := len $authors -}} {{- $pos := 0 -}}{{- $corr := false -}} {{- range $i, $a := $authors -}} {{- if and (reflect.IsMap $a) $a.highlight -}}{{- $pos = add $i 1 -}}{{- if $a.corresponding -}}{{- $corr = true -}}{{- end -}}{{- end -}} {{- end -}} {{- /* A missing author list and a one-name author list are different facts, and only the second is sole authorship. Neither is invented here: a work with nothing to go on is drawn with no mark and says so. */ -}} {{- if eq $n 0 -}} {{- warnf "authorship: %q has no `authors`; drawn without a mark" .Title -}} {{- else if eq $pos 0 -}} {{- warnf "authorship: %q has %d authors and none flagged `highlight: true`; drawn without a mark" .Title $n -}} {{- end -}} {{- /* conference items and reports are list-only, so those fall back to the works list filtered to the year, never to a publisher or a mirror */ -}} {{- $href := .RelPermalink -}} {{- if not $href -}}{{- $href = printf "%s?search=%d" $listURL .Date.Year -}}{{- end -}} {{- /* first and last are the positions that mean something; everything between them is the middle, however many there are */ -}} {{- $first := eq $pos 1 -}} {{- /* sole author is first author; last means someone else came before */ -}} {{- $last := and (gt $n 1) (eq $pos $n) -}} {{- $works = $works | append (dict "title" .Title "year" .Date.Year "pos" $pos "n" $n "href" $href "first" $first "last" $last "middle" (and (gt $pos 0) (not $first) (not $last)) "corr" $corr "open" (in $presented .Params.type) "kind" (i18n (printf "type_%s" .Params.type) | default .Params.type)) -}} {{- end -}} {{- with $works -}} {{- $years := slice -}} {{- range $works -}}{{- if not (in $years .year) -}}{{- $years = $years | append .year -}}{{- end -}}{{- end -}} {{- end -}}
