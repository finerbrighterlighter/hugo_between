{{- /* Research output, read as author lists. Replaces the separate output timeline, which plotted one mark per work by year. This does that — a column per work, grouped under its year, so the height of a year is still how much came out of it — and adds the thing a publication list never says: which of them were mine to lead. One slot per author, first at the top, my slot marked. Sole author is a single mark; third of eleven is a long column with the mark near the top. The old figure's one encoding is kept: filled for written output, open for presented. It is carried by my slot rather than the whole column. Works are spaced evenly inside a year rather than by date; four of them fall within months of each other in 2023 and would collide. HTML and CSS rather than SVG, so the year labels stay prose-sized text and the columns reflow — the old timeline had to be hidden below 700px. */ -}} {{- $enSite := site -}} {{- range hugo.Sites -}}{{- if eq .Language.Lang "en" -}}{{- $enSite = . -}}{{- end -}}{{- end -}} {{- $presented := slice "conference-speaking" "conference-poster" -}} {{- $pages := where (where $enSite.RegularPages "Section" "works") "Params.type" "ne" "under-review" -}} {{- $listURL := (partial "lang-link.html" "/works/").href -}} {{- $works := slice -}} {{- range (sort $pages "Date" "asc") -}} {{- $pos := 0 -}} {{- range $i, $a := .Params.authors -}}{{- if $a.highlight -}}{{- $pos = add $i 1 -}}{{- end -}}{{- end -}} {{- /* a work with no author list still counts as output: one slot, mine */ -}} {{- $n := len .Params.authors | default 1 -}} {{- if not $pos -}}{{- $pos = 1 -}}{{- $n = 1 -}}{{- end -}} {{- /* conference items and reports are list-only, so those fall back to the works list filtered to the year, never to a publisher or a mirror */ -}} {{- $href := .RelPermalink -}} {{- if not $href -}}{{- $href = printf "%s?search=%d" $listURL .Date.Year -}}{{- end -}} {{- $works = $works | append (dict "title" .Title "year" .Date.Year "pos" $pos "n" $n "href" $href "open" (in $presented .Params.type) "kind" (i18n (printf "type_%s" .Params.type) | default .Params.type)) -}} {{- end -}} {{- with $works -}} {{- $years := slice -}} {{- range $works -}}{{- if not (in $years .year) -}}{{- $years = $years | append .year -}}{{- end -}}{{- end -}} {{- /* the longest author list sets every year group's height, so the year labels sit on one line instead of stepping with each group */ -}} {{- $maxN := 1 -}} {{- range $works -}}{{- if gt .n $maxN -}}{{- $maxN = .n -}}{{- end -}}{{- end -}} {{- end -}}
