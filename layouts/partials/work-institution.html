{{- /* Institution mark at the top of a work page, the counterpart to the portrait that opens the home page. It says where the author was when the work was done, so it is opt-in per work via `institution: ` in front matter and is never inferred from the date. Keys are defined in data/institutions.yml. An unknown key is a content mistake worth surfacing, so it warns rather than failing the build; a work with no `institution` renders nothing. */ -}} {{- $key := .Params.institution -}} {{- with $key -}} {{- $inst := index site.Data.institutions . -}} {{- if not $inst -}} {{- errorf "work %q sets institution %q, which is not in data/institutions.yml. A mark asserts an affiliation, so a typo must fail rather than quietly drop it." $.File.Path . -}} {{- else -}} {{- if or (not $inst.source) (not $inst.rights) -}} {{- errorf "institution %q has no `source`/`rights` in data/institutions.yml. These marks are third-party trademarks outside the site's CC BY 4.0 licence; record where the file came from and on what basis it is used." . -}} {{- end -}} {{- $res := resources.Get $inst.logo -}} {{- if not $res -}} {{- errorf "institution %q points at %q, which is not under assets/" . $inst.logo -}} {{- else -}} {{- /* The mark is at most 108 CSS px (the portrait's clamp ceiling), so 2x is 216 and 3x is 324. PNG, not the site's usual WebP. Hugo's WebP output is lossy, and lossy WebP subsamples chroma, which is ruinous for these seals: fine gold lettering on a saturated blue ground is almost entirely chroma detail, and raising the quality setting does not help because the quantiser was never the problem. PNG is exact. Photographs keep using WebP; a flat-colour graphic behaves differently. `npm run marks:check` measures this if you want the numbers. */ -}} {{- $logo := $res.Resize "216x png Lanczos" -}} {{- $logo3x := $res.Resize "324x png Lanczos" -}}
{{- /* The design carries no caption, so the accessible name has to say what the mark means. Without it a logo at the top of a paper reads as the publisher or as a current endorsement, which is not the claim being made. */ -}} {{- $label := i18n "affiliation_at_time" (dict "name" $inst.name) -}}
{{- end -}} {{- end -}} {{- end -}}
